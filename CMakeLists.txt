cmake_minimum_required(VERSION 3.22.1)
project(MissionThunderboltPrototype VERSION 0.1 LANGUAGES C CXX)
include(CTest)
option(MT_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(MT_ENABLE_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use GCC or Clang for the combined ASan/UBSan checks")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_library(mt_core STATIC android/app/src/main/cpp/game.cpp android/app/src/main/cpp/api.cpp)
target_compile_features(mt_core PUBLIC cxx_std_17)
target_include_directories(mt_core PUBLIC android/app/src/main/cpp)
if(MSVC)
  target_compile_options(mt_core PRIVATE /W4 /WX)
else()
  target_compile_options(mt_core PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(mt_desktop_demo desktop/main.cpp)
target_link_libraries(mt_desktop_demo PRIVATE mt_core)
if(BUILD_TESTING)
  foreach(kind core api save fuzz manual_rules legacy_rng grid_layout grid_selector grid_fallback linked_record record_release selector_lifecycle a5_initialization glyph_preparation basic_cell_conversion direct_grid_conversion level_grid_conversion record_factory_tail record_template_factory record_followup finalized_template_factory supplied_record_placement factory_mode_variant factory_variant_draws child_record_prefix child_record_factory child_factory_context)
    add_executable(mt_${kind}_test tests/${kind}_test.cpp)
    target_link_libraries(mt_${kind}_test PRIVATE mt_core)
    if(MSVC)
      target_compile_options(mt_${kind}_test PRIVATE /UNDEBUG)
    else()
      target_compile_options(mt_${kind}_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${kind} COMMAND mt_${kind}_test)
  endforeach()
  add_executable(mt_c_abi_test tests/c_abi_test.c)
  target_link_libraries(mt_c_abi_test PRIVATE mt_core)
  add_test(NAME c_abi COMMAND mt_c_abi_test)
endif()

option(MT_BUILD_SDL_HOST "Build native SDL2 synthetic demo host" OFF)
if(MT_BUILD_SDL_HOST)
  find_package(SDL2 2.0.18 CONFIG REQUIRED)
  add_executable(mission-thunderbolt-demo desktop/sdl_main.cpp)
  target_link_libraries(mission-thunderbolt-demo PRIVATE mt_core SDL2::SDL2)
  if(MSVC)
    target_compile_options(mission-thunderbolt-demo PRIVATE /W4 /WX)
  else()
    target_compile_options(mission-thunderbolt-demo PRIVATE -Wall -Wextra -Werror)
  endif()
  if(APPLE)
    set_target_properties(mission-thunderbolt-demo PROPERTIES
      MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_GUI_IDENTIFIER org.missionthunderbolt.prototype
      MACOSX_BUNDLE_BUNDLE_NAME "Thunderbolt Research Demo"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1"
      MACOSX_BUNDLE_BUNDLE_VERSION "1")
  endif()
  install(TARGETS mission-thunderbolt-demo RUNTIME DESTINATION . BUNDLE DESTINATION .)
  if(BUILD_TESTING)
    add_test(NAME sdl_smoke COMMAND mission-thunderbolt-demo --smoke-test --save-dir "${CMAKE_CURRENT_BINARY_DIR}/smoke-state")
    set_tests_properties(sdl_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  endif()
endif()
